import java.lang.reflect.Proxy;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Collections;
import org.apache.hugegraph.HugeGraphSupplier;
import org.apache.hugegraph.backend.BackendColumn;
import org.apache.hugegraph.config.HugeConfig;
import org.apache.hugegraph.id.IdGenerator;
import org.apache.hugegraph.serializer.BinaryElementSerializer;
import org.apache.hugegraph.struct.schema.IndexLabel;
import org.apache.hugegraph.struct.schema.PropertyKey;
import org.apache.hugegraph.struct.schema.VertexLabel;
import org.apache.hugegraph.structure.BaseVertex;
import org.apache.hugegraph.structure.Index;
import org.apache.hugegraph.structure.builder.IndexBuilder;
import org.apache.hugegraph.type.HugeType;
import org.apache.hugegraph.type.define.DataType;
import org.apache.hugegraph.type.define.IndexType;

public class LegacyStoreIndexBuilderProbe {
    public static void main(String[] args) throws Exception {
        HugeConfig config = new HugeConfig(Collections.emptyMap());
        PropertyKey pkey = new PropertyKey(null, IdGenerator.of(17L), "name");
        pkey.dataType(DataType.TEXT);
        VertexLabel vlabel = new VertexLabel(null, IdGenerator.of(4L), "person");
        vlabel.properties(pkey.id());
        vlabel.ttl(60000L);
        HugeGraphSupplier graph = (HugeGraphSupplier) Proxy.newProxyInstance(
            HugeGraphSupplier.class.getClassLoader(), new Class<?>[]{HugeGraphSupplier.class},
            (proxy, method, values) -> {
                switch (method.getName()) {
                    case "configuration": return config;
                    case "propertyKey": return pkey;
                    case "vertexLabel":
                    case "vertexLabelOrNone": return vlabel;
                    case "now": return 1609459200000L;
                    case "name": return "legacy-fixture";
                    default: throw new AssertionError("Unexpected supplier call: " + method);
                }
            });
        IndexLabel ilabel = new IndexLabel(graph, IdGenerator.of(15L), "by-name");
        ilabel.indexType(IndexType.SECONDARY);
        ilabel.baseType(HugeType.VERTEX_LABEL);
        ilabel.baseValue(vlabel.id());
        ilabel.indexFields(pkey.id());
        String value = "long-field-".repeat(20);
        BaseVertex vertex = new BaseVertex(IdGenerator.of(1L), vlabel);
        vertex.addProperty(pkey, value);
        vertex.expiredTime(1609459260000L);
        IndexBuilder builder = new IndexBuilder(graph);
        Index index = builder.buildIndex(vertex, ilabel).get(0);
        BackendColumn column = new BinaryElementSerializer().writeIndex(index);
        String producer = Files.readString(Path.of(args[1]), StandardCharsets.UTF_8);
        String license = producer.substring(0, producer.indexOf("*/") + 2);
        String text = license.replace("/*", "#").replace("*/", "#").replace(" *", "#") + "\n" +
            "input.length=" + value.length() + "\n" +
            "store.builder.field.length=" + index.fieldValues().toString().length() + "\n" +
            "store.builder.field=" + index.fieldValues() + "\n" +
            "store.builder.key=" + hex(column.name) + "\n" +
            "store.builder.value=" + hex(column.value) + "\n";
        Files.writeString(Path.of(args[0]).resolve("index-builder-contract.properties"), text,
                          StandardCharsets.UTF_8);
        System.out.println("Original Store IndexBuilder: input " + value.length() +
                           " -> field " + index.fieldValues());
    }

    private static String hex(byte[] bytes) {
        StringBuilder text = new StringBuilder();
        for (byte value : bytes) { text.append(String.format("%02x", value & 0xff)); }
        return text.toString();
    }
}

import java.lang.reflect.Proxy;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.Map;
import org.apache.hugegraph.HugeGraphSupplier;
import org.apache.hugegraph.backend.BackendColumn;
import org.apache.hugegraph.id.Id;
import org.apache.hugegraph.id.EdgeId;
import org.apache.hugegraph.type.define.Directions;
import org.apache.hugegraph.id.IdGenerator;
import org.apache.hugegraph.serializer.BinaryElementSerializer;
import org.apache.hugegraph.struct.schema.IndexLabel;
import org.apache.hugegraph.struct.schema.PropertyKey;
import org.apache.hugegraph.struct.schema.VertexLabel;
import org.apache.hugegraph.structure.Index;
import org.apache.hugegraph.type.HugeType;
import org.apache.hugegraph.type.define.DataType;
import org.apache.hugegraph.type.define.IndexType;

public class LegacyStoreIndexFixtureWriter {
    private static final long EXPIRY = 1609459260000L;
    private static final Map<Long, IndexLabel> INDEXES = new HashMap<>();
    private static final Map<Long, PropertyKey> PROPERTIES = new HashMap<>();
    private static final Map<String, String> OUTPUT = new LinkedHashMap<>();

    public static void main(String[] args) throws Exception {
        VertexLabel label = new VertexLabel(null, IdGenerator.of(4L), "person");
        label.ttl(60000L);
        HugeGraphSupplier graph = (HugeGraphSupplier) Proxy.newProxyInstance(
            HugeGraphSupplier.class.getClassLoader(), new Class<?>[]{HugeGraphSupplier.class},
            (proxy, method, values) -> {
                switch (method.getName()) {
                    case "vertexLabel":
                    case "vertexLabelOrNone": return label;
                    case "indexLabel": return INDEXES.get(((Id) values[0]).asLong());
                    case "propertyKey": return PROPERTIES.get(((Id) values[0]).asLong());
                    case "now": return 1609459200000L;
                    case "name": return "legacy-fixture";
                    default: throw new AssertionError("Unexpected supplier call: " + method);
                }
            });
        PropertyKey intKey = new PropertyKey(graph, IdGenerator.of(1L), "age");
        intKey.dataType(DataType.INT);
        PROPERTIES.put(1L, intKey);
        PropertyKey longKey = new PropertyKey(graph, IdGenerator.of(16L), "weight");
        longKey.dataType(DataType.LONG);
        PROPERTIES.put(16L, longKey);
        BinaryElementSerializer serializer = new BinaryElementSerializer();
        System.out.println("Store index writer loaded from " + BinaryElementSerializer.class
            .getProtectionDomain().getCodeSource().getLocation());
        user("secondary", graph, serializer, 6L, IndexType.SECONDARY, "alice");
        user("int", graph, serializer, 7L, IndexType.RANGE_INT, 7);
        user("long", graph, serializer, 8L, IndexType.RANGE_LONG, 7L);
        Index system = new Index(graph, IndexLabel.label(HugeType.VERTEX), true);
        system.fieldValues(4L);
        system.elementIds(IdGenerator.of(1L), EXPIRY);
        row("system", serializer, graph, system);
        Index edgeSystem = new Index(graph, IndexLabel.label(HugeType.EDGE), true);
        edgeSystem.fieldValues(5L);
        edgeSystem.elementIds(new EdgeId(IdGenerator.of(1L), Directions.OUT,
                             IdGenerator.of(5L), IdGenerator.of(5L), "2026!x", IdGenerator.of(2L)), EXPIRY);
        row("system.edge", serializer, graph, edgeSystem);
        String source = Files.readString(Path.of(args[1]), StandardCharsets.UTF_8);
        String license = source.substring(0, source.indexOf("*/") + 2);
        StringBuilder text = new StringBuilder(license.replace("/*", "#")
            .replace("*/", "#").replace(" *", "#")).append('\n');
        OUTPUT.forEach((key, value) -> text.append(key).append('=').append(value).append('\n'));
        Files.writeString(Path.of(args[0]).resolve("store-index-rows.properties"), text,
                          StandardCharsets.UTF_8);
        System.out.println("Generated 5 original Store TTL index rows");
    }

    private static void user(String name, HugeGraphSupplier graph,
                              BinaryElementSerializer serializer, long id,
                              IndexType type, Object value) {
        IndexLabel label = new IndexLabel(graph, IdGenerator.of(id), "index-" + id);
        label.baseType(HugeType.VERTEX_LABEL);
        label.baseValue(IdGenerator.of(4L));
        label.indexType(type);
        label.indexFields(IdGenerator.of(type == IndexType.RANGE_LONG ? 16L : 1L));
        INDEXES.put(id, label);
        Index index = new Index(graph, label, true);
        index.fieldValues(value);
        index.elementIds(IdGenerator.of(1L), EXPIRY);
        row(name, serializer, graph, index);
    }

    private static void row(String name, BinaryElementSerializer serializer,
                             HugeGraphSupplier graph, Index index) {
        BackendColumn column = serializer.writeIndex(index);
        Index decoded = serializer.parseIndex(graph, column, null);
        if (decoded.expiredTime() != EXPIRY) {
            throw new AssertionError("Legacy Store reader TTL mismatch: " + name);
        }
        OUTPUT.put(name + ".name", hex(column.name));
        OUTPUT.put(name + ".value", hex(column.value));
        OUTPUT.put(name + ".expiry", String.valueOf(EXPIRY));
    }

    private static String hex(byte[] bytes) {
        StringBuilder text = new StringBuilder();
        for (byte value : bytes) {
            text.append(String.format("%02x", value & 0xff));
        }
        return text.toString();
    }
}
